     1	#!/usr/bin/env python3
     2	"""
     3	Video-QoS Simulator - DESiRED testbed (simplified DASH.js replacement)
     4	Uses real measured ping RTT through the 3-switch path as a congestion
     5	proxy, drives a buffer-level simulation, derives FPS/bitrate tier.
     6	Writes dash_<id>.txt in format: timestamp;bufferLocal;FPS;BitrateCalc
     7	"""
     8	import argparse
     9	import time
    10	import subprocess
    11	
    12	TIERS = [
    13	    (10.0, 30, 1040),
    14	    (3.0,  24,  490),
    15	    (0.0,  18,  140),
    16	]
    17	BUFFER_MAX = 64.0
    18	BUFFER_DRAIN_RATE = 1.0
    19	BUFFER_FILL_RATE = 1.5
    20	
    21	def choose_tier(buffer_level):
    22	    for min_buf, fps, bitrate in TIERS:
    23	        if buffer_level > min_buf:
    24	            return fps, bitrate
    25	    return 0, 0
    26	
    27	def measure_delay_once(dst_ip, timeout=1.0):
    28	    result = subprocess.run(
    29	        ["ping", "-c", "1", "-W", str(timeout), dst_ip],
    30	        capture_output=True, text=True
    31	    )
    32	    if result.returncode == 0 and "time=" in result.stdout:
    33	        time_str = result.stdout.split("time=")[1].split(" ")[0]
    34	        return float(time_str)
    35	    return None
    36	
    37	def main():
    38	    parser = argparse.ArgumentParser()
    39	    parser.add_argument("--dst", required=True)
    40	    parser.add_argument("--target-delay-ms", type=float, default=20.0)
    41	    parser.add_argument("--duration", type=int, default=60)
    42	    parser.add_argument("--out", required=True)
    43	    args = parser.parse_args()
    44	
    45	    buffer_level = 20.0
    46	    start = time.time()
    47	
    48	    with open(args.out, "a") as f:
    49	        while time.time() - start < args.duration:
    50	            second_start = time.time()
    51	            delay_ms = measure_delay_once(args.dst)
    52	
    53	            if delay_ms is None:
    54	                buffer_level = max(0.0, buffer_level - BUFFER_DRAIN_RATE * 2)
    55	            elif delay_ms <= args.target_delay_ms:
    56	                buffer_level = min(BUFFER_MAX, buffer_level + BUFFER_FILL_RATE)
    57	            else:
    58	                excess_ratio = delay_ms / args.target_delay_ms
    59	                buffer_level = max(0.0, buffer_level - BUFFER_DRAIN_RATE * excess_ratio)
    60	
    61	            fps, bitrate = choose_tier(buffer_level)
    62	            ts = str(int(time.time() * 1000))[:14]
    63	            line = f"{ts};{buffer_level:.2f};{fps};{bitrate}\n"
    64	            f.write(line)
    65	            f.flush()
    66	            print(f"[t={int(time.time()-start)}s] delay={delay_ms}ms buffer={buffer_level:.2f}s fps={fps} bitrate={bitrate}kbps")
    67	
    68	            elapsed = time.time() - second_start
    69	            if elapsed < 1.0:
    70	                time.sleep(1.0 - elapsed)
    71	
    72	if __name__ == "__main__":
    73	    main()

     3	scripts/validate_run.py
     4	
     5	Automated validator for DESiRED testbed experiment runs.
     6	Verifies data integrity, duration, telemetry pairing, network load,
     7	and controller execution for publishable experiments.
     8	
     9	Status outcomes:
    10	  VALID
    11	  INVALID_NETWORK_LOAD
    12	  INVALID_INT_CAPTURE
    13	  INVALID_DASH_DURATION
    14	  INVALID_CONTROLLER_EXIT
    15	  INVALID_CONFIG
    16	"""

    22	import re

    25	AGENT_LOG_DIR = Path("/home/p4/DESiRED/local_controller/agent_logs")

    27	def validate_run(run_id, expected_duration=700, max_dash_gap=8.5, min_int_ratio=0.85, is_controller=None):
    28	    details = []
    29	    
    30	    # Auto-detect if controller was expected if not explicitly specified
    31	    if is_controller is None:
    32	        if run_id.startswith("infer_") or run_id.startswith("train_") or run_id.startswith("online_") or run_id.startswith("rule_"):
    33	            is_controller = True
    34	        else:
    35	            is_controller = False
    36	
    37	    # 1. Check iperf client log
    38	    iperf_file = LOG_DIR / f"iperf_client_{run_id}.log"
    39	    if not iperf_file.exists():
    40	        details.append(f"Missing iperf client log: {iperf_file}")
    41	        return "INVALID_NETWORK_LOAD", details
    42	    
    43	    iperf_content = iperf_file.read_text(errors='ignore')
    44	    if "Connection refused" in iperf_content or "error - unable to connect" in iperf_content:
    45	        details.append("iperf connection refused or failed to connect.")
    46	        return "INVALID_NETWORK_LOAD", details
    47	    
    48	    if "iperf Done." not in iperf_content:
    49	        details.append("iperf log does not contain 'iperf Done.'. Run terminated early.")
    50	        return "INVALID_NETWORK_LOAD", details
    51	    
    52	    # Check for transfer volume
    53	    match_transfer = re.search(r'(\d+\.?\d*)\s+MBytes', iperf_content)
    54	    if not match_transfer and "KBytes" not in iperf_content:
    55	        details.append("iperf transfer data missing or malformed.")
    56	        return "INVALID_NETWORK_LOAD", details
    57	
    58	    # 2. Check sender log
    59	    sender_file = LOG_DIR / f"sender_{run_id}.log"
    60	    expected_probes = expected_duration
    61	    if sender_file.exists():
    62	        sender_lines = [l for l in sender_file.read_text(errors='ignore').splitlines() if l.strip()]
    63	        last_line = sender_lines[-1] if sender_lines else ""
    64	        match_sender = re.search(r'\[(\d+)/(\d+)\]', last_line)
    65	        if match_sender:
    66	            sent_count = int(match_sender.group(1))
    67	            total_count = int(match_sender.group(2))
    68	            # Use the number of probes actually SENT as the expected total.
    69	            # The sender is killed by cleanup_processes() at teardown, so
    70	            # sent_count < total_count is normal, not a failure.
    71	            expected_probes = sent_count if sent_count > 0 else total_count
    72	            if sent_count < int(total_count * 0.70):
    73	                # Only fail if sender barely started (< 70% sent)
    74	                details.append(f"Sender aborted very early: only {sent_count}/{total_count} probes sent.")
    75	                return "INVALID_INT_CAPTURE", details
    76	            elif sent_count < total_count:
    77	                details.append(f"Note: sender killed at teardown after {sent_count}/{total_count} probes (normal).")
    78	        else:
    79	            details.append(f"Sender log does not end with expected [N/N] probe line: {last_line}")
    80	    else:
    81	        # Note: some runs might have used stdout redirection differently
    82	        details.append(f"Warning: sender log {sender_file} not found; proceeding to check INT log.")
    83	
    84	    # 3. Check INT file
    85	    int_file = LOG_DIR / f"log_INT_{run_id}.txt"
    86	    if not int_file.exists():
    87	        details.append(f"Missing INT log: {int_file}")
    88	        return "INVALID_INT_CAPTURE", details
    89	    
    90	    int_lines = [l for l in int_file.read_text(errors='ignore').splitlines() if l.strip()]
    91	    int_count = len(int_lines)
    92	    min_probes = int(expected_probes * min_int_ratio)
    93	    if int_count < min_probes:
    94	        details.append(f"INT probe count too low: {int_count} received, expected at least {min_probes} ({min_int_ratio*100}% of {expected_probes}).")
    95	        return "INVALID_INT_CAPTURE", details
    96	
    97	    # 4. Check DASH file
    98	    dash_file = LOG_DIR / f"dash_{run_id}.txt"
    99	    if not dash_file.exists():
